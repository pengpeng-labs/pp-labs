# ---- SQL：建表（int/str 两列）----
sql CREATE TABLE notes (id int, msg str)
sql CREATE TABLE empty (a int)
sql CREATE TABLE dup (a int)

# ---- SQL：插入（数字 + 引号字符串）----
sql INSERT INTO notes (id,msg) VALUES (1,'hello')
sql INSERT INTO notes (id,msg) VALUES (2,'world')
sql INSERT INTO notes (id,msg) VALUES (3,'pp-db')

# ---- SQL：SELECT（全列 + WHERE 第 0 列）----
sql SELECT id,msg FROM notes
sql SELECT id,msg FROM notes WHERE id=2
sql SELECT id,msg FROM notes WHERE id=99

# ---- SQL：空表与错误路径 ----
sql SELECT id FROM empty
sql INSERT INTO ghost (id,msg) VALUES (1,'x')
sql CREATE TABLE broken
sql HELLO WORLD
